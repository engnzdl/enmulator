```rust
use std::path::PathBuf;

pub fn detect_sdk() -> Option<PathBuf> {
    // Check ANDROID_HOME / ANDROID_SDK_ROOT env vars
    for var in &["ANDROID_HOME", "ANDROID_SDK_ROOT"] {
        if let Ok(path) = std::env::var(var) {
            let p = PathBuf::from(path);
            if p.exists() {
                return Some(p);
            }
        }
    }

    // Platform defaults
    let home = dirs::home_dir()?;
    let candidates = [
        #[cfg(target_os = "macos")]
        home.join("Library/Android/sdk"),
        #[cfg(target_os = "linux")]
        home.join("Android/Sdk"),
        #[cfg(target_os = "windows")]
        home.join("AppData/Local/Android/Sdk"),
    ];

    for path in &candidates {
        if path.exists() {
            return Some(path.clone());
        }
    }
    None
}

pub fn get_avdmanager_path(sdk: &PathBuf) -> PathBuf {
    sdk.join("cmdline-tools/latest/bin/avdmanager")
}

pub fn get_emulator_path(sdk: &PathBuf) -> PathBuf {
    sdk.join("emulator/emulator")
}

pub fn get_adb_path(sdk: &PathBuf) -> PathBuf {
    sdk.join("platform-tools/adb")
}
```

```rust
pub fn load_config(config_path: &PathBuf) -> Config {

        let content = fs::read_to_string(config_path).unwrap_or_default();
        serde_json::from_str(&content).unwrap_or_default()
    } else {
        let config = Config::default();
        save_config(config_path, &config);
        config
    }
}
```

```rust
use serde::{Deserialize, Serialize};
use std::collections::HashMap;
use std::fs;
use std::path::PathBuf;
use std::sync::Mutex;

#[derive(Debug, Serialize, Deserialize, Clone)]
pub struct Device {
    pub id: String,
    pub display_name: String,
    pub avd_name: String,
    pub profile: Option<String>,
    pub api_level: u8,
    pub status: String,        // "stopped", "running", "error"
    pub port: u16,
    pub root_enabled: bool,
    pub adb_enabled: bool,
    pub created_at: String,
}

pub struct DeviceStore {
    pub devices: Mutex<HashMap<String, Device>>,
    pub devices_dir: PathBuf,
}

impl DeviceStore {
    pub fn new(devices_dir: PathBuf) -> Self {
        fs::create_dir_all(&devices_dir).ok();
        Self {
            devices: Mutex::new(HashMap::new()),
            devices_dir,
        }
    }

    pub fn list(&self) -> Vec<Device> {
        self.devices.lock().unwrap().values().cloned().collect()
    }

    pub fn get(&self, id: &str) -> Option<Device> {
        self.devices.lock().unwrap().get(id).cloned()
    }

    pub fn add(&self, device: Device) {
        self.devices.lock().unwrap().insert(device.id.clone(), device);
    }

    pub fn remove(&self, id: &str) {
        self.devices.lock().unwrap().remove(id);
        let device_dir = self.devices_dir.join(id);
        fs::remove_dir_all(device_dir).ok();
    }

    pub fn update_status(&self, id: &str, status: &str) {
        if let Some(dev) = self.devices.lock().unwrap().get_mut(id) {
            dev.status = status.to_string();
        }
    }
}
```

**Step 1: AVD manager module**

`src-tauri/src/avd_manager.rs`:
```rust
use crate::sdk;
use crate::device::Device;
use std::path::PathBuf;
use std::process::Command;

pub fn create_avd(
    sdk_path: &PathBuf,
    device_id: &str,
    display_name: &str,
    api_level: u8,
    device_definition: &str,
    devices_dir: &PathBuf,
) -> Result<Device, String> {
    let avd_name = format!("enmulator_{}", device_id);
    let avd_path = devices_dir.join(device_id);
    let avdmanager = sdk::get_avdmanager_path(sdk_path);

    // Determine system image package
    let image_tag = "google_apis"; // or "default"
    let package = format!(
        "system-images;android-{};{};x86_64",
        api_level, image_tag
    );

    let output = Command::new(&avdmanager)
        .args([
            "create", "avd",
            "--force",
            "--name", &avd_name,
            "--package", &package,
            "--tag", image_tag,
            "--abi", "x86_64",
            "--device", device_definition,
            "--path", avd_path.to_str().unwrap(),
        ])
        .output()
        .map_err(|e| format!("Failed to run avdmanager: {}", e))?;

    if !output.status.success() {
        let stderr = String::from_utf8_lossy(&output.stderr);
        return Err(format!("avdmanager failed: {}", stderr));
    }

    Ok(Device {
        id: device_id.to_string(),
        display_name: display_name.to_string(),
        avd_name,
        profile: Some(device_definition.to_string()),
        api_level,
        status: "stopped".to_string(),
        port: 5554, // Will be updated on start
        root_enabled: false,
        adb_enabled: true,
        created_at: chrono::Utc::now().to_rfc3339(),
    })
}
```

## Task 6: Emulator — Start/Stop Device

**Objective:** Process management for emulator instances

**Files:**
- Create: `src-tauri/src/emulator.rs`
- Modify: `src-tauri/src/main.rs`

**Step 1: Emulator module**

`src-tauri/src/emulator.rs`:
```rust
use crate::sdk;
use std::path::PathBuf;
use std::process::{Child, Command};
use std::sync::Mutex;
use std::collections::HashMap;

pub struct EmulatorStore {
    pub processes: Mutex<HashMap<String, Child>>,
    pub next_port: Mutex<u16>,
}

impl EmulatorStore {
    pub fn new() -> Self {
        Self {
            processes: Mutex::new(HashMap::new()),
            next_port: Mutex::new(5554),
        }
    }

    pub fn start(&self, sdk_path: &PathBuf, avd_name: &str, headless: bool) -> Result<u16, String> {
        let emulator = sdk::get_emulator_path(sdk_path);
        let port = {
            let mut p = self.next_port.lock().unwrap();
            let current = *p;
            *p += 2; // Emulator uses port pairs (5554/5555, 5556/5557...)
            current
        };

        let mut cmd = Command::new(&emulator);
        cmd.arg("-avd").arg(avd_name);
        cmd.arg("-port").arg(port.to_string());

        if headless {
            cmd.arg("-no-window");
        }

        let child = cmd.spawn()
            .map_err(|e| format!("Failed to start emulator: {}", e))?;

        self.processes.lock().unwrap().insert(avd_name.to_string(), child);
        Ok(port)
    }

    pub fn stop(&self, avd_name: &str) -> Result<(), String> {
        // Use ADB to shutdown gracefully
        let adb_path = sdk::get_adb_path(
            &crate::sdk::detect_sdk().ok_or("SDK not found")?
        );

        // Find the port for this AVD
        let port = {
            let procs = self.processes.lock().unwrap();
            // Kill via ADB emu kill
            let _ = Command::new(&adb_path)
                .args(["-s", &format!("emulator-{}", 5554), "emu", "kill"])
                .output();
        };

        // Force kill the process
        if let Some(mut child) = self.processes.lock().unwrap().remove(avd_name) {
            let _ = child.kill();
        }

        Ok(())
    }

    pub fn is_running(&self, avd_name: &str) -> bool {
        self.processes.lock().unwrap().contains_key(avd_name)
    }
}
```

```rust
use serde::{Deserialize, Serialize};
use std::path::PathBuf;
use std::fs;

#[derive(Debug, Serialize, Deserialize, Clone)]
pub struct FingerprintProfile {
    pub name: String,
    pub brand: String,
    pub model: String,
    pub manufacturer: String,
    pub device: String,
    pub fingerprint: String,
    pub dpi: u16,
    pub resolution_w: u16,
    pub resolution_h: u16,
}

pub fn list_profiles(profiles_dir: &PathBuf) -> Vec<FingerprintProfile> {
    let mut profiles = Vec::new();
    if let Ok(entries) = fs::read_dir(profiles_dir) {
        for entry in entries.flatten() {
            if let Ok(content) = fs::read_to_string(entry.path()) {
                if let Ok(profile) = serde_json::from_str::<FingerprintProfile>(&content) {
                    profiles.push(profile);
                }
            }
        }
    }
    profiles
}

pub fn save_profile(profiles_dir: &PathBuf, profile: &FingerprintProfile) {
    fs::create_dir_all(profiles_dir).ok();
    let path = profiles_dir.join(format!("{}.json", profile.name.replace(' ', "_").to_lowercase()));
    let content = serde_json::to_string_pretty(profile).unwrap();
    fs::write(path, content).ok();
}
```

```rust
use std::process::Command;
use std::path::PathBuf;

pub fn run_adb(adb_path: &PathBuf, device: &str, args: &[&str]) -> Result<String, String> {
    let output = Command::new(adb_path)
        .arg("-s").arg(device)
        .args(args)
        .output()
        .map_err(|e| format!("ADB error: {}", e))?;

    if output.status.success() {
        Ok(String::from_utf8_lossy(&output.stdout).to_string())
    } else {
        Err(String::from_utf8_lossy(&output.stderr).to_string())
    }
}

pub fn shell(adb_path: &PathBuf, device: &str, cmd: &str) -> Result<String, String> {
    run_adb(adb_path, device, &["shell", cmd])
}

pub fn setprop(adb_path: &PathBuf, device: &str, key: &str, value: &str) -> Result<(), String> {
    shell(adb_path, device, &format!("setprop {} {}", key, value)).map(|_| ())
}
```

```rust
use std::fs;

pub fn clone_device(store: &DeviceStore, source_id: &str, target_name: &str) -> Result<Device, String> {
    let source = store.get(source_id).ok_or("Source device not found")?;

    let target_id = target_name.to_lowercase().replace(' ', "_");
    let source_dir = store.devices_dir.join(source_id);
    let target_dir = store.devices_dir.join(&target_id);

    // Copy entire AVD directory
    fs::create_dir_all(&target_dir).map_err(|e| e.to_string())?;

    for entry in fs::read_dir(&source_dir).map_err(|e| e.to_string())? {
        let entry = entry.map_err(|e| e.to_string())?;
        let dest = target_dir.join(entry.file_name());
        if entry.path().is_dir() {
            // Skip snapshots for clone (they reference the original AVD name)
            if entry.file_name() != "snapshots" {
                copy_dir(&entry.path(), &dest)?;
            }
        } else {
            fs::copy(entry.path(), &dest).map_err(|e| e.to_string())?;
        }
    }

    let cloned = Device {
        id: target_id.clone(),
        display_name: target_name.to_string(),
        avd_name: format!("enmulator_{}", target_id),
        profile: source.profile.clone(),
        api_level: source.api_level,
        status: "stopped".to_string(),
        port: 0,
        root_enabled: source.root_enabled,
        adb_enabled: source.adb_enabled,
        created_at: chrono::Utc::now().to_rfc3339(),
    };

    store.add(cloned.clone());
    Ok(cloned)
}

fn copy_dir(src: &PathBuf, dst: &PathBuf) -> Result<(), String> {
    fs::create_dir_all(dst).map_err(|e| e.to_string())?;
    for entry in fs::read_dir(src).map_err(|e| e.to_string())? {
        let entry = entry.map_err(|e| e.to_string())?;
        let dest = dst.join(entry.file_name());
        if entry.path().is_dir() {
            copy_dir(&entry.path(), &dest)?;
        } else {
            fs::copy(entry.path(), &dest).map_err(|e| e.to_string())?;
        }
    }
    Ok(())
}
```
